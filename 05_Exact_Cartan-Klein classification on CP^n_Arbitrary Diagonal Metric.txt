from __future__ import annotations
from fractions import Fraction
from typing import List, Dict, Optional
import json, time, hashlib, math
from pathlib import Path


class IntegratedCartanKlein:
    """
    Exact Cartan-Klein classification on CP^n under an arbitrary diagonal metric.

    Base arithmetic is rational (Fraction): the Cartan inverse diagonal mu_k, the
    trace S, and the base volume are computed exactly; only the final gauge power
    (a real 1/4 root) and the Klein polynomial evaluation are floating point, and
    both are logged with the exact rational inputs that produced them.
    """

    KLEIN_SEQUENCE = [4, -2, -1, 3]   # Z_4-graded Klein opcode (exact integers)

    def __init__(self, n: int):
        self.n = n
        self.N = n + 1
        # exact Cartan inverse diagonal mu_k = (k+1)(N-k)/(N+1)
        self.mu: List[Fraction] = [
            Fraction((k + 1) * (self.N - k), self.N + 1) for k in range(self.N)
        ]
        self.S: Fraction = sum(self.mu)                      # exact trace
        # exact scale-invariant base volume = prod(mu_k / S)
        self.base_vol_exact: Fraction = Fraction(1, 1)
        for m in self.mu:
            self.base_vol_exact *= (m / self.S)
        # base gauge gamma_base = |base_vol|^{-1/4}  (real root, exact input logged)
        self.base_gauge: float = abs(float(self.base_vol_exact)) ** (-0.25)

    # ---- core Klein polynomial (Horner, exact integer coeffs) ----
    def _klein_poly(self, lam: float) -> float:
        coeffs = (self.KLEIN_SEQUENCE * ((self.N + 3) // 4))[:self.N]
        acc = 0.0
        for c in reversed(coeffs):
            acc = acc * lam + c
        return acc

    def default_chiral_metric(self) -> List[float]:
        """The polarized metric of vX1.0: leading half 2x, trailing half 1x."""
        half = self.N // 2
        return [2.0] * half + [1.0] * (self.N - half)

    def compute(self, p: int, g_diagonal: Optional[List[float]] = None) -> Dict:
        """
        Full conformal projection + chiral Casimir invariant under metric G.

        Returns a fully reproducible record: exact rational inputs, the per-state
        lambda / pi / state / log values, the signature, the polarized
        leading/trailing decomposition, and the metric-weighted invariant.
        """
        if g_diagonal is None:
            g_diagonal = self.default_chiral_metric()
        if len(g_diagonal) != self.N:
            raise ValueError(f"Metric G must have exactly N={self.N} diagonal entries.")

        gamma_p = (p ** (-self.N / 4.0)) * self.base_gauge

        per_state = []
        states = []
        for k, m in enumerate(self.mu):
            lam_exact = p * (m / self.S)          # exact Fraction
            lam = float(lam_exact)
            pi_val = self._klein_poly(lam)
            state = gamma_p * pi_val
            states.append(state)
            per_state.append({
                "k": k,
                "mu_k": str(m),                    # exact rational
                "lambda_exact": str(lam_exact),    # exact rational
                "lambda_float": lam,
                "pi_klein": pi_val,
                "state": state,
                "state_sign": 1 if state >= 0.0 else -1,
                "state_sq": state * state,
                "g_weight": g_diagonal[k],
                "weighted_sq": g_diagonal[k] * state * state,
                "ln_abs_state": math.log(abs(state)) if state != 0 else None,
            })

        signature = [d["state_sign"] for d in per_state]

        # polarized decomposition (general G): split at the half index
        half = self.N // 2
        leading_energy = sum(per_state[k]["weighted_sq"] for k in range(half))
        trailing_energy = sum(per_state[k]["weighted_sq"] for k in range(half, self.N))
        casimir = leading_energy + trailing_energy

        return {
            "manifold": f"CP^{self.n}",
            "n": self.n,
            "N": self.N,
            "prime": p,
            "klein_sequence": self.KLEIN_SEQUENCE,
            "exact_inputs": {
                "mu": [str(m) for m in self.mu],
                "S": str(self.S),
                "base_vol_exact": str(self.base_vol_exact),
                "base_gauge": self.base_gauge,
                "gamma_p": gamma_p,
            },
            "metric_G": list(g_diagonal),
            "per_state": per_state,
            "signature": signature,
            "leading_energy": leading_energy,
            "trailing_energy": trailing_energy,
            "casimir_invariant": casimir,
        }

            "casimir_invariant": "K_G(p) = sum_k G_k * state_k^2 = leading + trailing",
